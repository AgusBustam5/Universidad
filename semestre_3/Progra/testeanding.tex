\subsection*{1. Expresiones Regulares (Módulo \texttt{re})}
\begin{tabular}{lll}
\toprule
\textbf{Función} & \textbf{Uso principal} & \textbf{Contexto} \\
\midrule
\texttt{re.match(p, s)} & Busca al inicio & Validación simple \\
\texttt{re.fullmatch(p, s)} & Verifica todo el string & Validación estricta \\
\texttt{re.search(p, s)} & Primera coincidencia & Localizar patrón \\
\texttt{re.findall(p, s)} & Lista de coincidencias & Extracción de datos \\
\texttt{re.sub(p, r, s)} & Reemplazo & Limpieza de texto \\
\texttt{re.split(p, s)} & División & Procesar separadores \\
\bottomrule
\end{tabular}
